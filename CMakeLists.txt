cmake_minimum_required(VERSION 3.20)
project(PlatformChecks LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CHECKS_SANITIZERS "Enable address and undefined-behavior checks on Linux" OFF)

if(CHECKS_SANITIZERS AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The sanitizer configuration is supported on Linux only")
endif()

add_library(check_options INTERFACE)
if(MSVC)
    target_compile_options(check_options INTERFACE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(check_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
if(CHECKS_SANITIZERS)
    target_compile_options(check_options INTERFACE -O1 -g -fno-omit-frame-pointer
        -fsanitize=address,undefined -fno-sanitize-recover=all)
    target_link_options(check_options INTERFACE -fsanitize=address,undefined)
endif()

enable_testing()
function(add_check target)
    add_executable(${target} ${ARGN})
    target_link_libraries(${target} PRIVATE check_options)
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
endfunction()

add_check(portable_checks checks/portable/fragment.cpp checks/portable/checks.cpp)
if(NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
    add_test(NAME json_oracle COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/checks/portable/json_oracle.py $<TARGET_FILE:portable_checks>)
    set_tests_properties(json_oracle PROPERTIES TIMEOUT 30)
endif()

        _WIN32_WINNT=0x0602 WINVER=0x0602)
    add_check(win32_utf16 checks/windows/utf16.cpp)
    add_check(win32_display_config checks/windows/display_config.cpp)
    target_link_libraries(win32_display_config PRIVATE user32)
    add_check(win32_raw_input checks/windows/raw_input.cpp)
    target_link_libraries(win32_raw_input PRIVATE user32)
    add_check(win32_audio_endpoints checks/windows/audio_endpoints.cpp)
    target_link_libraries(win32_audio_endpoints PRIVATE ole32 uuid)
    add_check(win32_device_info checks/windows/device_info.cpp)
    target_link_libraries(win32_device_info PRIVATE setupapi)
    add_check(win32_sessions checks/windows/sessions.cpp)
    target_link_libraries(win32_sessions PRIVATE wtsapi32)
    add_check(win32_session_environment checks/session/environment.cpp)
    target_link_libraries(win32_session_environment PRIVATE wtsapi32 ole32 oleaut32 uuid user32 advapi32)
    add_executable(win32_window_calibration checks/window/calibration.cpp)
    target_link_libraries(win32_window_calibration PRIVATE check_options user32 gdi32 bcrypt)
endif()
