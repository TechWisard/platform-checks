cmake_minimum_required(VERSION 3.20)
project(RootDevicePropertyObservation LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This independent SetupAPI probe requires Windows")
endif()
add_executable(root_device_property probe.cpp)
target_compile_features(root_device_property PRIVATE cxx_std_17)
set_target_properties(root_device_property PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(root_device_property PRIVATE UNICODE _UNICODE
    WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602 WINVER=0x0602)
if(MSVC)
    target_compile_options(root_device_property PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(root_device_property PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(root_device_property PRIVATE -static)
endif()
target_link_libraries(root_device_property PRIVATE setupapi advapi32 ole32)
