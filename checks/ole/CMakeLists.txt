cmake_minimum_required(VERSION 3.20)
project(NativeOleHostProbe LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This runtime probe requires Windows")
endif()
add_executable(native_ole_host_probe probe.cpp ole_host.cpp)
target_compile_features(native_ole_host_probe PRIVATE cxx_std_17)
set_target_properties(native_ole_host_probe PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(native_ole_host_probe PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
if(MSVC)
    target_compile_options(native_ole_host_probe PRIVATE /W4 /WX /permissive- /utf-8)
    set_property(TARGET native_ole_host_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(native_ole_host_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(native_ole_host_probe PRIVATE -static)
endif()
target_link_libraries(native_ole_host_probe PRIVATE user32 gdi32 ole32 oleaut32 uuid)
