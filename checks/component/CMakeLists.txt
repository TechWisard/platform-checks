cmake_minimum_required(VERSION 3.20)
project(ComponentContractObservation LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This standalone COM contract probe requires Windows")
endif()
add_executable(component_contract probe.cpp)
target_compile_features(component_contract PRIVATE cxx_std_17)
set_target_properties(component_contract PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_definitions(component_contract PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602 WINVER=0x0602)
if(MSVC)
    target_compile_options(component_contract PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(component_contract PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(component_contract PRIVATE -static)
endif()
target_link_libraries(component_contract PRIVATE ole32 oleaut32 uuid)
