cmake_minimum_required(VERSION 3.20)
project(ChildSearchPreparation LANGUAGES CXX)
add_executable(child_search_receipt_tests receipt_tests.cpp)
target_compile_features(child_search_receipt_tests PRIVATE cxx_std_17)
set_target_properties(child_search_receipt_tests PROPERTIES CXX_EXTENSIONS OFF)
if(WIN32)
    add_executable(child_search_prepare prepare.cpp)
    target_compile_features(child_search_prepare PRIVATE cxx_std_17)
    set_target_properties(child_search_prepare PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_definitions(child_search_prepare PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    if(MSVC)
        target_compile_options(child_search_prepare PRIVATE /W4 /WX /permissive- /utf-8)
        set_property(TARGET child_search_prepare PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    else()
        target_compile_options(child_search_prepare PRIVATE -Wall -Wextra -Wpedantic -Werror)
        target_link_options(child_search_prepare PRIVATE -static -municode)
    endif()
    target_link_libraries(child_search_prepare PRIVATE user32 ole32 oleaut32 uuid wtsapi32 advapi32)
endif()
if(MSVC)
    target_compile_options(child_search_receipt_tests PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(child_search_receipt_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
