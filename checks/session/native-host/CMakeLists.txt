cmake_minimum_required(VERSION 3.20)
project(IndependentNativeHostBridge LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows is required; this target has no emulation path")
endif()
add_library(NativeHostBridge SHARED bridge.cpp ../../events/ole_host.cpp)
target_include_directories(NativeHostBridge PRIVATE ../../events)
set_target_properties(NativeHostBridge PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO PREFIX "")
target_compile_definitions(NativeHostBridge PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00 NOMINMAX)
target_link_libraries(NativeHostBridge PRIVATE ole32 oleaut32 user32 gdi32 uuid)
if(MSVC)
  target_compile_options(NativeHostBridge PRIVATE /W4 /WX /permissive- /EHsc)
  set_property(TARGET NativeHostBridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(NativeHostBridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(NativeHostBridge PRIVATE -static -static-libgcc -static-libstdc++)
endif()
