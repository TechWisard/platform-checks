cmake_minimum_required(VERSION 3.20)
project(FirstLogonPreparation LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This explicitly opted-in disposable-runner helper requires Windows")
endif()
add_executable(first_logon_prepare prepare.cpp)
target_compile_features(first_logon_prepare PRIVATE cxx_std_17)
set_target_properties(first_logon_prepare PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(first_logon_prepare PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
if(MSVC)
    target_compile_options(first_logon_prepare PRIVATE /W4 /WX /permissive- /utf-8)
    set_property(TARGET first_logon_prepare PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(first_logon_prepare PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(first_logon_prepare PRIVATE -static)
endif()
target_link_libraries(first_logon_prepare PRIVATE user32 ole32 oleaut32 uuid wtsapi32 dwmapi)
