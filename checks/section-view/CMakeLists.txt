cmake_minimum_required(VERSION 3.20)
project(SectionViewObservation LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This independent section probe requires Windows")
endif()
add_executable(section_view_observation probe.cpp)
target_compile_features(section_view_observation PRIVATE cxx_std_17)
set_target_properties(section_view_observation PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(section_view_observation PRIVATE UNICODE _UNICODE
    WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602 WINVER=0x0602)
if(MSVC)
    target_compile_options(section_view_observation PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(section_view_observation PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion)
    target_link_options(section_view_observation PRIVATE -static)
endif()
