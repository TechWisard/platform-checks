cmake_minimum_required(VERSION 3.20)
project(RuntimeSyntheticFixtures LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "These two synthetic fixtures require Windows.")
endif()
add_executable(runtime_fixture_console SyntheticConsole.cpp)
add_executable(runtime_fixture_window WIN32 SyntheticWindow.cpp)
target_link_libraries(runtime_fixture_window PRIVATE user32)
foreach(target runtime_fixture_console runtime_fixture_window)
  target_compile_features(${target} PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(${target} PRIVATE -static)
  endif()
endforeach()
if(MINGW)
  target_link_options(runtime_fixture_window PRIVATE -municode)
endif()
