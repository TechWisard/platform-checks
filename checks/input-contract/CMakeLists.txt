cmake_minimum_required(VERSION 3.20)
project(native_input_contract LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This probe requires Windows")
endif()
add_executable(input_contract probe.cpp)
target_compile_features(input_contract PRIVATE cxx_std_17)
target_compile_definitions(input_contract PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(input_contract PRIVATE ole32 oleaut32 uuid)
if(MSVC)
  set_property(TARGET input_contract PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(input_contract PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
else()
  target_compile_options(input_contract PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(input_contract PRIVATE -static-libgcc -static-libstdc++)
endif()
