cmake_minimum_required(VERSION 3.20)
project(NativeEventDeliveryProbe LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This runtime probe requires Windows")
endif()
add_executable(native_event_delivery_probe probe.cpp ole_host.cpp)
target_compile_features(native_event_delivery_probe PRIVATE cxx_std_17)
set_target_properties(native_event_delivery_probe PROPERTIES
    CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_definitions(native_event_delivery_probe PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
if(MSVC)
    target_compile_options(native_event_delivery_probe PRIVATE /W4 /WX /permissive- /utf-8)
    set_property(TARGET native_event_delivery_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(native_event_delivery_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(native_event_delivery_probe PRIVATE -static)
endif()
target_link_libraries(native_event_delivery_probe PRIVATE user32 gdi32 ole32 oleaut32 uuid advapi32 ws2_32)
