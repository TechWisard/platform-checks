cmake_minimum_required(VERSION 3.20)
project(PresentationObservation LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This read-only probe requires Windows")
endif()
add_executable(presentation_observation probe.cpp)
target_compile_features(presentation_observation PRIVATE cxx_std_17)
set_target_properties(presentation_observation PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(presentation_observation PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0602 WINVER=0x0602)
if(MSVC)
    target_compile_options(presentation_observation PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(presentation_observation PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(presentation_observation PRIVATE -static)
endif()
target_link_libraries(presentation_observation PRIVATE user32 ole32 oleaut32 uuid wtsapi32)
